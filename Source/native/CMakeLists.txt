cmake_minimum_required(VERSION 3.28)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" SCOUT_VERSION LIMIT_COUNT 1)
project(prime_gap_scout VERSION ${SCOUT_VERSION} LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.4 REQUIRED)
find_package(Threads REQUIRED)
find_library(PGN_CUDA_DEVICE_RUNTIME NAMES cudadevrt HINTS
 "${CUDAToolkit_LIBRARY_DIR}" "${CUDAToolkit_TARGET_DIR}/lib64"
 "${CUDAToolkit_TARGET_DIR}/lib/x64" "${CUDAToolkit_TARGET_DIR}/lib" REQUIRED)
if(WIN32)
 find_file(PGN_CUDART_RUNTIME NAMES cudart64_${CUDAToolkit_VERSION_MAJOR}.dll
  HINTS "${CUDAToolkit_BIN_DIR}" "${CUDAToolkit_TARGET_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
 install(FILES "${PGN_CUDART_RUNTIME}" DESTINATION bin)
else()
 find_library(PGN_CUDART_RUNTIME NAMES cudart HINTS "${CUDAToolkit_LIBRARY_DIR}"
  "${CUDAToolkit_TARGET_DIR}/lib64" "${CUDAToolkit_TARGET_DIR}/lib" REQUIRED)
 get_filename_component(PGN_CUDART_REAL "${PGN_CUDART_RUNTIME}" REALPATH)
 install(FILES "${PGN_CUDART_REAL}" DESTINATION lib RENAME libcudart.so.${CUDAToolkit_VERSION_MAJOR})
endif()
option(PGN_ENABLE_LTO "Retain validated device LTO; OFF is explicitly unqualified" ON)
set(PGN_RELEASE_ARCHS "auto" CACHE STRING "auto,86,120 or semicolon-separated subset")
set(PGN_RELEASE_GEOMETRIES "8;16;32;64" CACHE STRING "Descriptor K profiles to build")
if(NOT PGN_ENABLE_LTO)
 message(WARNING "Device LTO disabled explicitly: this build has no inherited performance qualification")
endif()
execute_process(COMMAND "${CMAKE_CUDA_COMPILER}" --list-gpu-code OUTPUT_VARIABLE PGN_SUPPORTED_CODE RESULT_VARIABLE PGN_CODE_QUERY TIMEOUT 15)
if(NOT PGN_CODE_QUERY EQUAL 0)
 message(FATAL_ERROR "Cannot query nvcc architectures; refusing an assumed target list")
endif()
if(PGN_RELEASE_ARCHS STREQUAL "auto")
 set(PGN_RELEASE_ARCHS "86")
 if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 12.8 AND PGN_SUPPORTED_CODE MATCHES "sm_120")
  list(APPEND PGN_RELEASE_ARCHS 120)
 else()
  message(STATUS "Skipping sm120: Blackwell requires CUDA12.8+ with sm_120 support; building sm86 only")
 endif()
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 foreach(PGN_INCLUDE IN LISTS CUDAToolkit_INCLUDE_DIRS)
  if(EXISTS "${PGN_INCLUDE}/crt/host_config.h")
   file(READ "${PGN_INCLUDE}/crt/host_config.h" PGN_HOST_CONFIG)
   string(REGEX MATCH "__GNUC__[ \t]*>[ \t]*([0-9]+)" PGN_GNU_GUARD "${PGN_HOST_CONFIG}")
   if(PGN_GNU_GUARD)
    set(PGN_MAX_GNU "${CMAKE_MATCH_1}")
    string(REGEX MATCH "^[0-9]+" PGN_ACTUAL_GNU "${CMAKE_CXX_COMPILER_VERSION}")
    if(PGN_ACTUAL_GNU GREATER PGN_MAX_GNU)
     message(FATAL_ERROR "GCC${PGN_ACTUAL_GNU} exceeds toolkit host_config limit${PGN_MAX_GNU}; select a compatible host compiler. No silent override.")
    endif()
   endif()
   break()
  endif()
 endforeach()
endif()
foreach(ARCH IN LISTS PGN_RELEASE_ARCHS)
 if(NOT ARCH MATCHES "^(86|120)$")
  message(FATAL_ERROR "Only release architectures86 and120 are supported")
 endif()
 if(ARCH STREQUAL "120" AND CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
  message(FATAL_ERROR "Explicit sm120 request requires CUDA12.8 or newer")
 endif()
 if(NOT PGN_SUPPORTED_CODE MATCHES "sm_${ARCH}([^0-9]|$)")
  message(FATAL_ERROR "Selected nvcc does not support requested sm_${ARCH}")
 endif()
 foreach(K IN LISTS PGN_RELEASE_GEOMETRIES)
  if(NOT K MATCHES "^(8|16|32|64)$")
   message(FATAL_ERROR "K must be8,16,32 or64")
  endif()
  set(TARGET pgn68-worker-sm${ARCH}-k${K})
  add_executable(${TARGET} search_worker.cu)
  set_target_properties(${TARGET} PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17 CUDA_ARCHITECTURES OFF
   CUDA_SEPARABLE_COMPILATION ON CUDA_RESOLVE_DEVICE_SYMBOLS ON CUDA_RUNTIME_LIBRARY Shared
   INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
  target_compile_definitions(${TARGET} PRIVATE PGN_RELEASE_BUILD_ARCH=${ARCH} PGN_RELEASE_ENABLE_PUBLIC_PERIG=0
   PGN_DESCRIPTOR_LANES=2 PGN_DESCRIPTOR_BLOCKS=${K} PGN_EULER96_LAZY PGN_PRESIEVE_GROUPS=1
   PGN_MARK_FULL_FAST PGN_MARK_BANK_ROTATE PGN_FUSED_SIEVE PGN_EULER96 PGN_EULER96_FIXED_HIGH10 PGN_VERIFY_GAPS)
  target_compile_options(${TARGET} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_${ARCH};-O3;-lineinfo;--threads=1;--ptxas-options=-v>")
  target_link_options(${TARGET} PRIVATE "$<DEVICE_LINK:-arch=sm_${ARCH}>")
  if(PGN_ENABLE_LTO)
   target_compile_options(${TARGET} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-dlto>")
   target_link_options(${TARGET} PRIVATE "$<DEVICE_LINK:-dlto>")
  endif()
  # Conditional graph control uses the device Runtime API: retain RDC and
  # cudadevrt even for an explicit --no-lto build.
  target_link_libraries(${TARGET} PRIVATE CUDA::cudart "${PGN_CUDA_DEVICE_RUNTIME}" Threads::Threads ${CMAKE_DL_LIBS})
  if(MSVC)
   set_property(TARGET ${TARGET} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
   target_compile_options(${TARGET} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/MT;-Xcompiler=/utf-8>")
   target_link_options(${TARGET} PRIVATE "LINKER:/MANIFEST:EMBED" "LINKER:/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/worker.manifest")
  endif()
  install(TARGETS ${TARGET} RUNTIME DESTINATION bin)
 endforeach()
endforeach()
